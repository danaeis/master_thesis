% جدول‌های بلندی که باید میان صفحه‌ها شکسته شوند (فصل کارهای پیشین).
\usepackage{longtable}


% -------------------------------------------------------
%  ستون میان‌چین با پهنای ثابت برای جدول‌هایی که از صفحه بیرون می‌زنند
%  استفاده: \begin{tabular}{|p{3cm}|C{1.5cm}|...}
% -------------------------------------------------------
\usepackage{array}
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}

\newcommand{\OPT}{\ensuremath{\mathop{\mathrm{OPT}}}}
\newcommand{\APX}{\ensuremath{\mathop{\mathrm{APX}}}}
\newcommand{\ALG}{\ensuremath{\mathop{\mathrm{ALG}}}}


% -------------------------------------------------------
%  عدد فارسی در متن راست‌به‌چپ
%  اعداد چندبخشی (اعشار، هزارگان، بازه) در متن فارسی
%  بدون جهت‌دهی صریح وارونه چاپ می‌شوند؛ این دستور
%  ترتیب صحیح را با حفظ ارقام فارسی تضمین می‌کند.
% -------------------------------------------------------

\newcommand{\عدد}[1]{\LRE{#1}}

% ---------------------------------------------------------------
%  نشانگرهای موقت — پیش از تحویل نهایی باید همه حذف شوند.
%  برای یافتن همه:  grep -nE "نیازمندمقدار|نامطمئن|بایدتکمیل" chapters/*.tex
% ---------------------------------------------------------------
\usepackage{xcolor}

% قرمز: عددی که باید از متن اصلی مقاله استخراج و جایگزین شود.
\newcommand{\نیازمندمقدار}{\textcolor{red}{\textbf{[عدد؟]}}}

% قرمز: متنی که مقدارش نامطمئن است (عدد یا ادعایی که تأیید نشده).
\newcommand{\نامطمئن}[1]{\textcolor{red}{#1}}

% نارنجی: مطلبی که باید بعداً تکمیل یا بازنویسی شود.
\newcommand{\بایدتکمیل}[1]{\textcolor{orange}{#1}}

% آبی: یادداشت داخلی برای نویسنده (در نسخه‌ی نهایی حذف شود).
\newcommand{\یادداشت}[1]{\textcolor{blue}{\footnotesize [یادداشت: #1]}}

% تیک و ضربدر برای ستون «کد باز» در جدول‌ها.
\newcommand{\xmark}{\ensuremath{\times}}
